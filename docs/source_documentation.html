"""Run the golf-shot prediction workflow and display its results.

Runs data preparation, linear regression and the combined neural network.
Each child script prints its progress and saves evaluation graphs as PNGs.

For inline graphs in Google Colab, run in a notebook cell:
    %run "/content/MMA3001-Project/Main.py"

Using !python also runs the workflow, but cannot render inline notebook
images from the separate Python process. PNGs are still saved to model outputs.
"""

from pathlib import Path
import subprocess
import sys


ROOT = Path(__file__).resolve().parent
CODE_FOLDER = ROOT / "Code files"
OUTPUT_FOLDER = ROOT / "model outputs"

# List expected graph filenames in the order they should appear.
GRAPH_FILES = (
    "linear_regression_training.png",
    "linear_regression_validation.png",
    "nn_combined_side_total.png",
    "nn_combined_total.png",
)


def run_script(filename):
    """Execute one project script and stream its printed output and errors."""
    script = CODE_FOLDER / filename
    if not script.is_file():
        raise FileNotFoundError(f"Cannot find project script: {script}")

    print(f"\n=== Running {filename} ===", flush=True)
    # No output capture: results and errors appear directly in Colab.
    subprocess.run([sys.executable, "-u", str(script)], cwd=ROOT, check=True)


def show_graphs():
    """Display saved figures inline in IPython, or print their file paths."""
    print("\n=== Model evaluation graphs ===", flush=True)
    try:
        from IPython import get_ipython
        from IPython.display import Image, display
        in_notebook = get_ipython() is not None and "ipykernel" in sys.modules
    except ImportError:
        in_notebook = False

    for filename in GRAPH_FILES:
        image_path = OUTPUT_FOLDER / filename
        if not image_path.is_file():
            print(f"Graph not found: {image_path}", flush=True)
            continue

        print(f"Graph: {filename}", flush=True)
        if in_notebook:
            # %run executes inside the notebook kernel, enabling inline display.
            display(Image(filename=str(image_path)))
        else:
            print(f"Saved at: {image_path}", flush=True)


def main():
    """Prepare data, evaluate both models, and show the saved graphs."""
    OUTPUT_FOLDER.mkdir(parents=True, exist_ok=True)

    # Delete only known old graph files, avoiding stale results after reruns.
    for filename in GRAPH_FILES:
        (OUTPUT_FOLDER / filename).unlink(missing_ok=True)

    run_script("Data_processing.py")
    run_script("linear_regression_side_total.py")
    run_script("NN_Combined.py")
    show_graphs()


if __name__ == "__main__":
    main()

"""Train and evaluate a multi-output neural network for golf-shot outcomes.

Uses TrackMan-derived swing and ball-strike measurements to predict Side Total
and Total. The script loads prepared training and validation CSV files,
standardises input features using training data statistics, trains a PyTorch
neural network, reports mean-squared-error losses, and plots actual versus
predicted values with MAE and R-squared metrics.

Run Data_processing.py first. Dataset paths are resolved relative to this
script, not the process's current working directory.
"""

import os
import sys
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import mean_absolute_error, r2_score
from pathlib import Path

# The network maps the 15 input measurements to two outputs: Side Total and Total.
class GolfNeuralNet(nn.Module):
    """Fully connected network predicting Side Total and Total."""

    def __init__(self, input_dim):
        """Initialise the network layers.

        Args:
            input_dim (int): Number of input features supplied to the network.
        """
        super(GolfNeuralNet, self).__init__()
        # Hidden fully connected layers learn nonlinear combinations of input measurements.
        self.fc1 = nn.Linear(input_dim, 500)
        self.relu1 = nn.ReLU()
        self.fc2 = nn.Linear(500, 500)
        self.relu2 = nn.ReLU()
        self.fc3 = nn.Linear(500, 2)  # Output layer predicting 2 values (Side Total, Total)

    def forward(self, x):
        """Compute the two target predictions for a batch of input features.

        Args:
            x (torch.Tensor): Input tensor shaped (batch_size, input_dim).

        Returns:
            torch.Tensor: Predictions shaped (batch_size, 2), ordered as
            Side Total then Total.
        """
        out = self.fc1(x)
        out = self.relu1(out)
        out = self.fc2(out)
        out = self.relu2(out)
        out = self.fc3(out)
        return out

# Resolve data paths relative to this script so the current working directory
# and absolute repository location do not need to be hard-coded.
REPO_ROOT = Path(__file__).resolve().parent.parent
DATA_FOLDER = REPO_ROOT / "data files"

# Explicit input and target names are shared by training and source-level tests.
features = [
    'Club Speed', 'Attack Angle', 'Club Path', 'Low Point', 'Swing Plane',
    'Swing Direction', 'Dyn. Loft', 'Face Angle', 'Face To Path', 'Ball Speed',
    'Smash Factor', 'Launch Angle', 'Launch Direction', 'Spin Rate', 'Spin Axis'
]
targets = ['Side Total', 'Total']


def main():
    """Load datasets, train the combined model and plot evaluation results.

    Called only when this script is run directly, so importing GolfNeuralNet
    for documentation or testing does not trigger training or plotting.
    """
    # Load the Training and Validation datasets.
    train_file = DATA_FOLDER / "Training data.csv"
    val_file = DATA_FOLDER / "Validation data.csv"

    if not train_file.is_file():
        raise FileNotFoundError(
            f"Training dataset not found: {train_file}. "
            "Run Data_processing.py first."
        )
    if not val_file.is_file():
        raise FileNotFoundError(
            f"Validation dataset not found: {val_file}. "
            "Run Data_processing.py first."
        )

    train_df = pd.read_csv(train_file)
    val_df = pd.read_csv(val_file)

    # Split into features (X) and targets (y) using available, verified columns
    X_train = train_df[features]
    y_train = train_df[targets]
    X_val = val_df[features]
    y_val = val_df[targets]

    # Normalize features
    # Fit scaling parameters on training data only; reuse them unchanged for validation.
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_val_scaled = scaler.transform(X_val)

    # Convert to PyTorch Tensors
    X_train_tensor = torch.tensor(X_train_scaled, dtype=torch.float32)
    y_train_tensor = torch.tensor(y_train.values, dtype=torch.float32)
    X_val_tensor = torch.tensor(X_val_scaled, dtype=torch.float32)
    y_val_tensor = torch.tensor(y_val.values, dtype=torch.float32)

    # Instantiate Model, Loss Function, and Optimizer
    model = GolfNeuralNet(len(features))
    # MSE penalises squared differences between predicted and measured target values.
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=0.01)

    # Train the Model
    print("\nTraining Multi-Output Neural Network Model...")
    epochs = 150
    train_losses = []
    val_losses = []

    # One epoch is a complete optimisation pass over the training dataset.
    for epoch in range(1, epochs + 1):
        model.train()
        optimizer.zero_grad()
        outputs = model(X_train_tensor)
        loss = criterion(outputs, y_train_tensor)
        # Backpropagate the loss gradients, then update the model weights.
        loss.backward()
        optimizer.step()

        # Validation data measures generalisation; it is not used for gradient updates.
        # Evaluate on Validation set
        model.eval()
        with torch.no_grad():
            val_outputs = model(X_val_tensor)
            val_loss = criterion(val_outputs, y_val_tensor)

        train_losses.append(loss.item())
        val_losses.append(val_loss.item())

        if epoch == 1 or epoch % 25 == 0:
            print(f"Epoch {epoch:03d}/{epochs} | Train Loss (MSE): {loss.item():.4f} | Val Loss (MSE): {val_loss.item():.4f}")

    # Model evaluation
    model.eval()
    # Disable gradient tracking during final prediction to reduce overhead.
    with torch.no_grad():
        train_preds_np = model(X_train_tensor).numpy()
        val_preds_np = model(X_val_tensor).numpy()

    # Plot actual vs predicted values side-by-side
    for idx, target_name in enumerate(targets):
        plt.figure(figsize=(12, 5))

        # Compare predictions with known targets; the dashed diagonal indicates perfect predictions.
        # Train plot
        plt.subplot(1, 2, 1)
        tr_true = y_train.values[:, idx]
        tr_pred = train_preds_np[:, idx]
        mae_tr = mean_absolute_error(tr_true, tr_pred)
        r2_tr = r2_score(tr_true, tr_pred)
        plt.scatter(tr_true, tr_pred, alpha=0.3, color='blue')
        plt.plot([tr_true.min(), tr_true.max()], [tr_true.min(), tr_true.max()], 'r--', lw=2)
        plt.title(f'Train: Actual vs Pred ({target_name})\nMAE: {mae_tr:.2f}, R²: {r2_tr:.4f}')
        plt.xlabel('Actual')
        plt.ylabel('Predicted')

        # Val plot
        plt.subplot(1, 2, 2)
        v_true = y_val.values[:, idx]
        v_pred = val_preds_np[:, idx]
        mae_val = mean_absolute_error(v_true, v_pred)
        r2_val = r2_score(v_true, v_pred)
        plt.scatter(v_true, v_pred, alpha=0.3, color='green')
        plt.plot([v_true.min(), v_true.max()], [v_true.min(), v_true.max()], 'r--', lw=2)
        plt.title(f'Val: Actual vs Pred ({target_name})\nMAE: {mae_val:.2f}, R²: {r2_val:.4f}')
        plt.xlabel('Actual')
        plt.ylabel('Predicted')

        plt.tight_layout()
        output_dir = REPO_ROOT / "model outputs"
        output_dir.mkdir(parents=True, exist_ok=True)
        figure_path = output_dir / f"nn_combined_{target_name.lower().replace(' ', '_')}.png"
        plt.savefig(figure_path, dpi=150, bbox_inches="tight")
        print(f"Saved graph: {figure_path}")
        plt.close()


if __name__ == "__main__":
    main()

def test_features_exclude_targets(script_tree):

def test_scaler_uses_training_data_only(script_tree):
    """Check the script's scaler calls and demonstrate their expected behaviour."""
    # Inspect actual source calls: fitting must use X_train, not X_val.
    calls = [
        node for node in ast.walk(script_tree)
        if isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute)
        and isinstance(node.func.value, ast.Name)
        and node.func.value.id == "scaler"
    ]
    fitting_calls = [
        call for call in calls if call.func.attr in ("fit", "fit_transform")
    ]
    assert len(fitting_calls) == 1, "Scaler should be fitted exactly once"
    assert len(fitting_calls[0].args) >= 1
    assert isinstance(fitting_calls[0].args[0], ast.Name)
    assert fitting_calls[0].args[0].id == "X_train"

    validation_transforms = [
        call for call in calls
        if call.func.attr == "transform"
        and call.args and isinstance(call.args[0], ast.Name)
        and call.args[0].id == "X_val"
    ]
    assert validation_transforms, "Validation features must use scaler.transform"

    # Small synthetic example: validation data must not change training means.
    train = np.array([[0.0, 10.0], [2.0, 20.0], [4.0, 30.0]])
    validation = np.array([[100.0, 200.0]])
    scaler = StandardScaler()
    scaled_train = scaler.fit_transform(train)
    training_means = scaler.mean_.copy()
    scaled_validation = scaler.transform(validation)

    np.testing.assert_allclose(scaler.mean_, training_means)
    np.testing.assert_allclose(scaled_train.mean(axis=0), [0.0, 0.0], atol=1e-12)
    assert np.isfinite(scaled_validation).all()


def test_one_training_step_updates_weights(network_class):
    """A single optimisation step should compute a finite loss and change weights."""
    torch.manual_seed(42)  # Make the test repeatable.
    model = network_class(input_dim=15)
    optimizer = torch.optim.Adam(model.parameters(), lr=0.001)
    criterion = torch.nn.MSELoss()

    # Synthetic inputs/targets keep the test fast and independent of CSV files.
    inputs = torch.randn(8, 15)
    targets = torch.randn(8, 2)
    original_weights = [parameter.detach().clone() for parameter in model.parameters()]

    model.train()
    optimizer.zero_grad()
    predictions = model(inputs)
    loss = criterion(predictions, targets)
    assert torch.isfinite(loss), "Training loss must be finite"

    loss.backward()
    optimizer.step()

    assert any(
        not torch.equal(before, after)
        for before, after in zip(original_weights, model.parameters())
    ), "No network parameters changed during training"

"""Establish a linear-regression baseline for predicting golf-shot Side Total.

The script selects a configurable number of input features using SelectKBest
with f_regression, fits a linear regression model, reports MAE and R-squared
for training and validation data, and plots actual versus predicted values.
It expects the training and validation CSV files in the repository's data
files directory; run Data_processing.py first.
"""

import pandas as pd
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_absolute_error, r2_score
from sklearn.feature_selection import SelectKBest, f_regression
import os
from pathlib import Path

def run_side_total_regression_with_feature_selection(
    df_train_path,
    df_test_path,
    feature_columns,
    k_features=5,
):
    """Fit and evaluate a feature-selected linear regression model for Side Total.

    Args:
        df_train_path (str or pathlib.Path): Path to the training CSV file.
        df_test_path (str or pathlib.Path): Path to the validation/test CSV file.
        feature_columns (list[str]): Explicit names of candidate predictor
            columns. Target columns must not be included.
        k_features (int): Number of candidate features to retain, ranked using
            univariate linear-regression tests.

    Returns:
        tuple: Training MAE and R-squared, training actual and predicted values,
        validation MAE and R-squared, validation actual and predicted values,
        and the list of selected feature names.

    Raises:
        RuntimeError: If the training or validation CSV cannot be read.
        ValueError: If required columns are missing or no usable numeric rows remain.
    """
    # Load training and validation data, preserving informative error messages.
    try:
        df_train_fs = pd.read_csv(df_train_path)
        df_test_fs = pd.read_csv(df_test_path)
    except Exception as exc:
        raise RuntimeError(f"Could not load training or validation data: {exc}") from exc

    target_col = "Side Total"
    if not feature_columns:
        raise ValueError("Provide at least one candidate feature column.")
    if target_col in feature_columns or "Total" in feature_columns:
        raise ValueError("Target columns must not be included among the input features.")

    # Validate both datasets explicitly so column-order changes cannot alter inputs.
    required_columns = list(feature_columns) + [target_col]
    for dataset_name, dataset in (("training", df_train_fs), ("validation", df_test_fs)):
        missing = [column for column in required_columns if column not in dataset.columns]
        if missing:
            raise ValueError(f"The {dataset_name} dataset is missing columns: {missing}")

    # Convert selected predictors and the target to numeric values.
    X_train_fs = df_train_fs.loc[:, feature_columns].apply(pd.to_numeric, errors="coerce")
    y_train_fs = pd.to_numeric(df_train_fs[target_col], errors="coerce")
    X_test_fs_all = df_test_fs.loc[:, feature_columns].apply(pd.to_numeric, errors="coerce")
    y_test_fs_target = pd.to_numeric(df_test_fs[target_col], errors="coerce")

    # Drop rows with missing/non-numeric values while keeping predictors and targets aligned.
    aligned_train_fs = pd.concat([X_train_fs, y_train_fs.rename(target_col)], axis=1).dropna()
    aligned_test_fs = pd.concat([X_test_fs_all, y_test_fs_target.rename(target_col)], axis=1).dropna()

    X_train_fs = aligned_train_fs.loc[:, feature_columns]
    y_train_fs = aligned_train_fs[target_col]
    X_test_fs_all = aligned_test_fs.loc[:, feature_columns]
    y_test_side_total = aligned_test_fs[target_col]

    if aligned_train_fs.empty or aligned_test_fs.empty:
        raise ValueError("No complete numeric rows remain in the training or validation dataset.")

    if not 1 <= k_features <= len(feature_columns):
        raise ValueError(
            f"k_features must be between 1 and {len(feature_columns)}; got {k_features}."
        )

    # Feature Selection for 'Side Total'
    # Rank candidate features using their individual linear relationship with Side Total.
    # Fit the selector on training data only to avoid using validation outcomes for selection.
    selector_side_total = SelectKBest(f_regression, k=k_features)
    selector_side_total.fit(X_train_fs, y_train_fs)
    selected_features_side_total_list = X_train_fs.columns[selector_side_total.get_support()].tolist()

    # Filter X_train and X_test to include only the selected features for 'Side Total'
    X_train_side_total_selected = X_train_fs[selected_features_side_total_list]
    X_test_side_total_selected = X_test_fs_all[selected_features_side_total_list]

    # Train Linear Regression model for 'Side Total'
    model_side_total = LinearRegression()
    # Fit the regression coefficients using only the selected training features.
    model_side_total.fit(X_train_side_total_selected, y_train_fs)

    # Make predictions on Training Data
    y_pred_train = model_side_total.predict(X_train_side_total_selected)
    # MAE measures average absolute prediction error; R² measures explained variation.
    mae_train = mean_absolute_error(y_train_fs, y_pred_train)
    r2_train = r2_score(y_train_fs, y_pred_train)

    # Make predictions on Validation/Testing Data
    y_pred_validation = model_side_total.predict(X_test_side_total_selected)
    mae_validation = mean_absolute_error(y_test_side_total, y_pred_validation)
    r2_validation = r2_score(y_test_side_total, y_pred_validation)

    return (
        mae_train, r2_train, y_train_fs, y_pred_train,
        mae_validation, r2_validation, y_test_side_total, y_pred_validation,
        selected_features_side_total_list
    )

"""Prepare TrackMan golf-shot data for regression and neural-network modelling.

Reads the original CSV from the repository's data files directory, skips the
measurement-units row, removes unused columns, keeps Driver shots, removes rows
with missing values, and writes the cleaned dataset plus separate training
and validation CSV files.

Run this script before either modelling script. Paths are resolved relative to
this file, so the current working directory does not need to be the repository
root.
"""

# pandas reads and writes the CSV datasets; pathlib builds portable file paths.
import pandas as pd
from pathlib import Path

# Resolve paths from this script's location, not the current working directory.
# This assumes the script is stored in the repository's "Code files" directory.
REPO_ROOT = Path(__file__).resolve().parent.parent
data_folder = REPO_ROOT / "data files"
input_file = data_folder / "data.csv"
output_file = data_folder / "cleaned_data.csv"

# 1. Load the data, skipping the units row
# The second CSV row contains units rather than observations, so it is skipped.
df = pd.read_csv(input_file, skiprows=[1])
print(f"Rows initially loaded: {len(df)}")

# 2. Remove columns not required for the model
# These columns contain missing values and are excluded from the cleaned dataset.
columns_to_drop = [
    "Ball",
    "Dynamic Lie",
    "Impact Offset",
    "Impact Height",
]
# Drop only the specified columns; pandas raises an error if a listed column is absent.
df = df.drop(columns=columns_to_drop)

# 3. Keep only Driver shots
df = df[df["Club"] == "Driver"].copy()
print(f"Rows after filtering for Driver: {len(df)}")

# 4. Remove rows containing any missing values
# Keep only complete observations so later models receive numeric values for every field.
df = df.dropna()
print(f"Rows after removing missing values: {len(df)}")

# 5. Save the cleaned dataset
df.to_csv(output_file, index=False)

print(f"Cleaned data saved to: {output_file}")
print(f"Final number of rows: {len(df)}")
print(f"Final number of columns: {len(df.columns)}")


# 6. Shuffle once with a fixed seed, then create training and validation datasets.
# A fixed seed makes the split reproducible whenever the input data is unchanged.
shuffled_df = df.sample(frac=1, random_state=42).reset_index(drop=True)
num_parts = 2
chunk_size = len(shuffled_df) // num_parts

# Part 1: Training Data
train_part = shuffled_df.iloc[0:chunk_size]
train_output_file = data_folder / "Training data.csv"
train_part.to_csv(train_output_file, index=False)
print(f"Training data saved to: {train_output_file} ({len(train_part)} rows)")

# Part 2: Validation Data
# Any remainder row goes into validation when the row count is odd.
val_part = shuffled_df.iloc[chunk_size:]
val_output_file = data_folder / "Validation data.csv"
val_part.to_csv(val_output_file, index=False)
print(f"Validation data saved to: {val_output_file} ({len(val_part)} rows)")
